/* Переносит форму GetCourse (блок «Форма» этой страницы) в место .gc-form-slot.
   Сначала берутся обычные блоки формы, потом всплывающие; пустой обычный блок скрывается. */
(function () {
  function moveForms() {
    var root = document.querySelector('.lp-ii3');
    if (!root) return;
    var slots = root.querySelectorAll('.gc-form-slot:not(.is-filled)');
    if (!slots.length) return;
    var normal = [], popup = [];
    Array.prototype.forEach.call(
      document.querySelectorAll('form.lt-form, form.lt-normal-form'),
      function (f) {
        if (root.contains(f)) return;
        (f.closest('.lt-modal-block') ? popup : normal).push(f);
      }
    );
    var forms = normal.concat(popup);
    for (var i = 0; i < slots.length && i < forms.length; i++) {
      var block = forms[i].closest('.lt-block');
      var isPopup = !!forms[i].closest('.lt-modal-block');
      slots[i].appendChild(forms[i]);
      slots[i].classList.add('is-filled');
      if (block && !isPopup) block.style.display = 'none';
    }
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', moveForms);
  else moveForms();
  window.addEventListener('load', moveForms);
})();

/* Дата и время эфира — то же правило, что на текущей странице:
   ближайшая 10-минутная отметка вперёд по Москве, с переносом часа и суток. */
(function () {
  var MONTHS = ['января','февраля','марта','апреля','мая','июня',
                'июля','августа','сентября','октября','ноября','декабря'];
  var DIM = [31,28,31,30,31,30,31,31,30,31,30,31];
  function pad(n){ return String(n).length < 2 ? '0' + n : '' + n; }
  function render(){
    var root = document.querySelector('.lp-ii3');
    if (!root) return;
    var v = {};
    new Intl.DateTimeFormat('ru-RU', {timeZone:'Europe/Moscow', day:'numeric', month:'numeric',
      year:'numeric', hour:'2-digit', minute:'2-digit', hour12:false})
      .formatToParts(new Date()).forEach(function(p){ if (p.type !== 'literal') v[p.type] = p.value; });
    var day = Number(v.day), month = Number(v.month) - 1, year = Number(v.year);
    var hour = Number(v.hour), minute = Number(v.minute);
    var rounded = Math.ceil((minute + 1) / 10) * 10;
    if (rounded >= 60) { rounded -= 60; hour += 1; }
    if (hour >= 24) {
      hour -= 24;
      var dim = DIM.slice();
      if ((year % 4 === 0 && year % 100 !== 0) || year % 400 === 0) dim[1] = 29;
      if (day + 1 > dim[month]) { day = 1; month = month + 1 > 11 ? 0 : month + 1; } else { day += 1; }
    }
    var time = pad(hour) + ':' + pad(rounded);
    Array.prototype.forEach.call(root.querySelectorAll('.lp-time'), function (el) { el.textContent = time; });
    Array.prototype.forEach.call(root.querySelectorAll('.lp-date'), function (el) {
      el.textContent = day + ' ' + MONTHS[month];
    });
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', render);
  else render();
  setInterval(render, 30000);
})();

/* UTM-метки из адреса страницы → скрытые поля формы GetCourse.
   Заполняем после загрузки и ещё раз перед отправкой: иначе значения стираются,
   пока GetCourse готовит форму. */
(function () {
  var FIELDS = {utm_source: 979724, utm_medium: 979725, utm_campaign: 979726, utm_content: 979727, utm_term: 979728};
  function param(name) {
    var m = location.search.match(new RegExp('[?&]' + name + '=([^&#]*)'));
    if (!m) return '';
    try { return decodeURIComponent(m[1].replace(/\+/g, ' ')); } catch (e) { return m[1]; }
  }
  function fillUtm() {
    Object.keys(FIELDS).forEach(function (key) {
      var value = param(key);
      if (!value) return;
      Array.prototype.forEach.call(
        document.querySelectorAll('input[name="formParams[userCustomFields][' + FIELDS[key] + ']"]'),
        function (input) { input.value = value; }
      );
    });
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', fillUtm);
  else fillUtm();
  window.addEventListener('load', function () { fillUtm(); setTimeout(fillUtm, 1500); });
  document.addEventListener('click', function (e) { if (e.target.closest && e.target.closest('form')) fillUtm(); }, true);
  document.addEventListener('submit', fillUtm, true);
})();
